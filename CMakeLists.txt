cmake_minimum_required(VERSION 3.16)
project(Inheritance_Lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ACCOUNT_SOURCES BankAccount.cpp CheckingAccount.cpp SavingsAccount.cpp)
add_executable(bank_accounts main.cpp ${ACCOUNT_SOURCES})
enable_testing()

add_test(NAME BankAccountTests COMMAND account_tests)
